function majorityVote(numbers) {
  const filtered = numbers.filter((n) => typeof n === 'number' && !isNaN(n));
  if (filtered.length === 0) return 0;
  const counts = {};
  filtered.forEach((n) => {
    counts[n] = (counts[n] || 0) + 1;
  });
  let best = filtered[0];
  let bestCount = 0;
  Object.entries(counts).forEach(([n, c]) => {
    const num = parseInt(n, 10);
    if (c > bestCount) {
      bestCount = c;
      best = num;
    }
  });
  return best;
}

function openingBlack(imgData) {
    const eroded = erodeBlack(imgData);
    const opened = dilateBlack(eroded);
    return opened;
  }
  function erodeBlack(imgData) {
    const { width, height, data } = imgData;
    const out = new ImageData(width, height);
    out.data.set(data);
    for (let y = 1; y < height - 1; y++) {
      for (let x = 1; x < width - 1; x++) {
        const i = (y * width + x) * 4;
        if (data[i] !== 0) continue;
        let hasWhite = false;
        for (let dy = -1; dy <= 1 && !hasWhite; dy++)
          for (let dx = -1; dx <= 1 && !hasWhite; dx++) {
            const ni = ((y + dy) * width + (x + dx)) * 4;
            if (data[ni] === 255) hasWhite = true;
          }
        if (hasWhite) out.data[i] = out.data[i+1] = out.data[i+2] = 255;
      }
    }
    return out;
  }

// 검은 픽셀을 3x3으로 한 겹 확장 (획 두껍게, 2→7 오인식 감소)
function dilateBlack(imgData) {
    const { width, height, data } = imgData;
    const out = new ImageData(width, height);
    out.data.set(data);
  
    for (let y = 1; y < height - 1; y++) {
      for (let x = 1; x < width - 1; x++) {
        const i = (y * width + x) * 4;
        if (data[i] === 0 && data[i + 1] === 0 && data[i + 2] === 0) continue;
  
        let hasBlack = false;
        for (let dy = -1; dy <= 1 && !hasBlack; dy++) {
          for (let dx = -1; dx <= 1 && !hasBlack; dx++) {
            const ni = ((y + dy) * width + (x + dx)) * 4;
            if (data[ni] === 0 && data[ni + 1] === 0 && data[ni + 2] === 0) hasBlack = true;
          }
        }
        if (hasBlack) {
          out.data[i] = out.data[i + 1] = out.data[i + 2] = 0;
        } else {
          out.data[i] = out.data[i + 1] = out.data[i + 2] = 255;
        }
      }
    }
    return out;
  }

// ---------- 1) 템플릿 이미지 로드 (컴포넌트 상단 또는 Test 밖에 함수 정의) ----------
function loadTemplateImage(src) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error('템플릿 이미지 로드 실패'));
    img.src = src;
  });
}

// ---------- 2) 그레이스케일 픽셀 배열로 변환 (R 채널만 사용) ----------
function imageToGrayData(img, width, height) {
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext('2d');
  ctx.drawImage(img, 0, 0, width, height);
  const imgData = ctx.getImageData(0, 0, width, height);
  const data = imgData.data;
  const gray = new Uint8Array(width * height);
  for (let i = 0; i < gray.length; i++) {
    gray[i] = (data[i * 4] + data[i * 4 + 1] + data[i * 4 + 2]) / 3;
  }
  return gray;
}

// ---------- 3) 단순 템플릿 매칭 (SSD 최소 위치 반환) ----------
function matchTemplate(sceneData, sceneW, sceneH, templateData, templateW, templateH, maxScore = Infinity) {
  let bestX = 0, bestY = 0, bestScore = Infinity;
  const maxX = sceneW - templateW;
  const maxY = sceneH - templateH;
  if (maxX <= 0 || maxY <= 0) return null;

  for (let y = 0; y <= maxY; y += 2) {
    for (let x = 0; x <= maxX; x += 2) {
      let ssd = 0;
      for (let ty = 0; ty < templateH; ty++) {
        for (let tx = 0; tx < templateW; tx++) {
          const si = ((y + ty) * sceneW + (x + tx)) * 4;
          const ti = ty * templateW + tx;
          const sr = (sceneData[si] + sceneData[si + 1] + sceneData[si + 2]) / 3;
          const diff = sr - templateData[ti];
          ssd += diff * diff;
        }
      }
      if (ssd < bestScore) {
        bestScore = ssd;
        bestX = x;
        bestY = y;
      }
    }
  }
  if (bestScore > maxScore) return null;   // ← 이 줄 추가 (return 바로 위)
  return { x: bestX, y: bestY, w: templateW, h: templateH };
}

function drawMatchBox(ctx, x, y, w, h, style = { stroke: '#00ff00', lineWidth: 3 }) {
  ctx.strokeStyle = style.stroke;
  ctx.lineWidth = style.lineWidth;
  ctx.strokeRect(x, y, w, h);
}


// OCR 직전 2배 확대 (얇은 숫자 2→7 오인식 감소)
function canvasToDataURLUpscaled(canvas, scale = 2) {
  const w = canvas.width;
  const h = canvas.height;
  const temp = document.createElement('canvas');
  temp.width = w * scale;
  temp.height = h * scale;
  const ctx = temp.getContext('2d');
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, temp.width, temp.height);
  ctx.drawImage(canvas, 0, 0, w, h, 0, 0, w * scale, h * scale);
  return temp.toDataURL('image/png');
}
