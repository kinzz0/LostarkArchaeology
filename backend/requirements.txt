#   pip uninstall -y torch torchvision
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu126
# (드라이버 CUDA 11.8 이면 cu118 URL 사용). 확인:
#   python -c "import torch; print('cuda:', torch.cuda.is_available(), torch.version.cuda)"
# 코드는 torch.cuda.is_available() 이면 자동 GPU(YOLO). CPU 강제만 .env 에 YOLO_DEVICE=cpu

#   .\scripts\install_paddle.ps1 -Cuda cu126    # GPU (CUDA 12.6 휠, 문서 기준 3.3.0)
#   .\scripts\install_paddle.ps1 -Cuda cu118    # CUDA 11.8 휠
#   .\scripts\install_paddle.ps1 -Cuda cpu
